    (cx : rust::ref (rust::mut' context))
    : rust_caught (poll t)
    =
    $'std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| std::future::Future::poll((!f).as_mut(), !cx)))'
    : rust_caught (poll t)

:= run_synchronously_rust \* t. (a : async t) : t =
    := waker = current_thread () |> thread_waker
    := context = waker |> rust::to_ref |> context_of_waker
    := future = a |> future_of_async
    future |> rust::to_mut
    context |> rust::to_mut
    := result = mut (optionm'::none' ())
    rust::loop 0 \ () =>
        := polled = poll_once (rust::to_ref_mut future) (rust::to_ref_mut context) |> poll_option
        result <- polled
        := _ =
            ? optionm'::is_none' polled ?>
                thread_park ()
                !-
            |:
                rust::break ()
                !+
        ()
    *result |> optionm'::unwrap

:= caught_poll \* t. (caught : rust_caught (poll t)) : poll (resultm::result' t exn) =
    ? rust_caught_ok caught ?>
        := polled = caught |> rust_caught_value |> poll_option
        ? optionm'::is_none' polled ?> poll_pending () |: polled |> optionm'::unwrap |> resultm::ok'' |> poll_ready
    |: caught |> rust_caught_payload |> rust_panic_text |> rust_exn_of_text |> resultm::err |> poll_ready
