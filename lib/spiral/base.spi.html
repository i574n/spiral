:= rust_catch_unwind \* t. (fn : () -> t) : rust_caught t =
    := fn = dyn fn
    $'std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| (!fn)()))' : rust_caught t

:= rust_quietly \* t. (catch : () -> t) : t =
    ? rust_panic_aborts () ?> catch ()
    |:
        := previous = rust_panic_take_hook ()
        rust_panic_silent_hook () |> rust_panic_set_hook
        := caught = catch ()
        rust_panic_set_hook previous
        caught

:= rust_caught_ok \* t. (caught : rust_caught t) : bool =
    $'!caught.is_ok()' : bool

:= rust_caught_value \* t. (caught : rust_caught t) : t =
    $'!caught.unwrap()' : t

:= rust_caught_payload \* t. (caught : rust_caught t) : rust_panic_payload =
    $'!caught.err().unwrap()' : rust_panic_payload

:= rust_panic_holds_string (payload : rust_panic_payload) : bool =
    $'!payload.downcast_ref::<String>().is_some()' : bool

:= rust_panic_holds_str (payload : rust_panic_payload) : bool =
    $'!payload.downcast_ref::<&str>().is_some()' : bool

:= rust_panic_string (payload : rust_panic_payload) : string =
    $'Rc::<str>::from(!payload.downcast_ref::<String>().unwrap().as_str())' : string

:= rust_panic_str (payload : rust_panic_payload) : string =
    $'Rc::<str>::from(*!payload.downcast_ref::<&str>().unwrap())' : string

:= rust_panic_text (payload : rust_panic_payload) : string =
    ? rust_panic_holds_string payload ?> rust_panic_string payload
    |? rust_panic_holds_str payload ?> rust_panic_str payload
    |: "panic"

:= rust_exn_of_text (text : string) : exn =
    $'!text' : exn

:= catch_unwind_rust \* t. (fn : () -> t) (ex_fn : exn -> t) : t =
    rust_quietly \ () =>
        := caught = rust_catch_unwind fn
        ? rust_caught_ok caught ?> rust_caught_value caught
        |: caught |> rust_caught_payload |> rust_panic_text |> rust_exn_of_text |> ex_fn

    $'!attempt[1]' : bool

:= lua_attempt_value \* t. (attempt : lua_attempt) : t =
    $'!attempt[2]' : t

:= lua_attempt_error (attempt : lua_attempt) : exn =
    $'!attempt[2]' : exn

:= try \* t. (fn : () -> t) (ex_fn : exn -> option t) : option t =
    := some x : option t = Some x
    := some = dyn some
    := fn = dyn fn
    := ex_fn = dyn ex_fn
    backend_switch {
        Gleam = \ () =>
            fn () |> some
        Lua = \ () =>
            := attempt = lua_pcall fn
            ? lua_attempt_ok attempt ?> attempt |> lua_attempt_value |> some
            |: attempt |> lua_attempt_error |> ex_fn
        Fsharp = \ () =>
            $'(try !fn () |> !some with ex -> !ex_fn ex)' : option t
        Python = \ () =>
            global """
                def spiral_try(fn, some, ex_fn):
                    try:
                        return some(fn())
                    except Exception as e:
                        return ex_fn(e)
                """
            $'spiral_try(!fn, !some, !ex_fn)' : option t
        Rust = \ () =>
            catch_unwind_rust (\ () => some (fn ())) ex_fn
        TypeScript = \ () =>
            $'(() => { try { return !some(!fn()); } catch (ex) { return !ex_fn(ex); } })()' : option t
        Zig = \ () =>
            fn () |> some
        C = \ () =>
            fn () |> some
    }
