:= lua_math_random (low : i32) (high : i32) : i32 = $'math.random(!low, !high)' : i32

:= native_rng_state () : () =
    global "static SPIRAL_NATIVE_RNG: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);"

:= native_rng_load () : u64 =
    native_rng_state ()
    $'SPIRAL_NATIVE_RNG.load(std::sync::atomic::Ordering::Relaxed)' : u64

:= native_rng_store (x : u64) : () = $'SPIRAL_NATIVE_RNG.store(!x, std::sync::atomic::Ordering::Relaxed)' : ()

:= xorshift (x : u64) : u64 =
    := x = x ^^^ (x <<< 13i32)
    := x = x ^^^ (x >>> 7i32)
    x ^^^ (x <<< 17i32)

:= native_rng_seed () : u64 =
    := a, _ = guid::rust_random_pair ()
    a ||| 1u64

:= native_rng_next () : u64 =
    := current = native_rng_load ()
    := next = xorshift (? current = 0u64 ?> native_rng_seed () |: current)
    native_rng_store next
    next

:= random_in_span (x : u64) (lo : i64) (hi : i64) : i64 =
    := span = hi - lo
    ? span <= 0i64 ?> lo |: lo + convert::i64 (x % convert::u64 span)

:= typescript_random_u53 () : u64 = $'BigInt(Math.floor(Math.random() * 9007199254740992))' : u64

:= native_random_between (lo : i64) (hi : i64) : i64 = random_in_span (native_rng_next ()) lo hi

:= process_counter_add (name : string) (n : u64) : u64 =
    run_target \|
        | Rust => \ () =>
            global """
                fn spiral_process_counter_add(name: &str, n: u64) -> u64 {
                    use std::collections::HashMap;
                    use std::sync::Mutex;
                    static COUNTERS: Mutex<Option<HashMap<String, u64>>> = Mutex::new(None);
                    let mut guard = COUNTERS.lock().unwrap();
                    let total = guard.get_or_insert_with(HashMap::new).entry(name.to_string()).or_insert(0);
                    *total += n;
                    *total
                }
                """
            $'spiral_process_counter_add(&*!name, !n)' : u64
        | _ => \ () => failwith "process_counter_add: Rust only; on the BEAM use an otp_actor counter"

:= process_counter_value (name : string) : u64 = process_counter_add name 0u64
