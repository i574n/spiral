:= write_line_c \* t. (obj : t) : () =
    ~>
        ?: t ?|
        | string => $'printf("%s\\n", (!obj)->ptr)' : ()
        | i32 => $'printf("%d\\n", !obj)' : ()
        | i64 => $'printf("%lld\\n", (long long)(!obj))' : ()
        | u32 => $'printf("%u\\n", !obj)' : ()
        | u64 => $'printf("%llu\\n", (unsigned long long)(!obj))' : ()
        | _ => $'backend_switch_has_no_C_arm_in_lib_spiral' : ()

:= write_line_gleam \* t. (obj : t) : () =
    global "import gleam/io"
    ? (~> ?: t ?| string => !+ | _ => !-) ?> $'io.println(!obj)' : ()
    |? (~> ?: t ?| () => !+ | _ => !-) ?> $'io.println("")' : ()
    |:
        global "import gleam/int"
        $'io.println(int.to_string(!obj))' : ()

:= write_wasm \* t. (obj : t) : () =
    ~>
        ?: t ?|
        | string => $'(call $print_str !obj)' : ()
        | i32 => $'(call $print_i64 (i64.extend_i32_s !obj))' : ()
        | i64 => $'(call $print_i64 !obj)' : ()
        | u32 => $'(call $print_u64 (i64.extend_i32_u !obj))' : ()
        | u64 => $'(call $print_u64 !obj)' : ()
        | () => $'' : ()
        | _ => $'backend_switch_has_no_Wasm_arm_in_lib_spiral' : ()

:= write_line_bend \* t. (x : t) : () =
    ~>
        ?: t ?|
        | string => !!!!Printf("%s\n", x)
        | i32 => !!!!Printf("%d\n", x)
        | _ => $'backend_switch_has_no_Bend_arm_in_lib_spiral' : ()

:= write_line_lua \* t. (obj : t) : () =
    ~>
        ?: t ?|
        | i64 => $'print(spiral_w64.tostring(!obj, true))' : ()
        | u64 => $'print(spiral_w64.tostring(!obj, false))' : ()
        | _ => $'print(!obj)' : ()

:= write_line \* t. (obj : t) : () =
    backend_switch {
        Gleam = \ () => write_line_gleam obj
        Lua = \ () => write_line_lua obj
        Fsharp = \ () =>
            \ () => obj |> $'System.Console.WriteLine'
            |> exec_unit
        Python = \ () => $'print(!obj)' : ()
        Rust = \ () => $'println\!("{}", !obj)' : ()
        TypeScript = \ () => $'process.stdout.write(String(!obj) + "\\n")' : ()
        Zig = \ () =>
            $'spiralPrintAny(!obj)' : ()
            $'spiralPrint(&[_]u8{10})' : ()
        Lean = \ () => $'IO.println (toString !obj)' : ()
        C = \ () => write_line_c obj
        Pascal = \ () => $'Writeln(!obj)' : ()
        Kotlin = fun () => $'run { System.out.write((!obj.toString() + "\\n").toByteArray(Charsets.UTF_8)); \
            System.out.flush() }' : ()
        Swift = \ () => $'print(!obj)' : ()
        Bend = \ () => write_line_bend obj
        Wasm = \ () =>
            write_wasm obj
            $'(call $print_newline)' : ()
    }

:= write_error_line \* t. (obj : t) : () =
    backend_switch {
        Gleam = \ () =>
            global "import gleam/io"
            $'io.println_error(!obj)' : ()
        Lua = \ () => $'io.stderr:write(tostring(!obj) .. "\\n")' : ()
        Fsharp = \ () =>
            \ () => obj |> $'System.Console.Error.WriteLine'
            |> exec_unit
        Python = \ () =>
            global "import sys"
            $'print(!obj, file=sys.stderr)' : ()
        Rust = \ () => $'eprintln\!("{}", !obj)' : ()
        TypeScript = \ () => $'process.stderr.write(String(!obj) + "\\n")' : ()
        Lean = \ () => $'IO.eprintln (toString !obj)' : ()
        Kotlin = fun () => $'run { System.err.write((!obj.toString() + "\\n").toByteArray(Charsets.UTF_8)); \
            System.err.flush() }' : ()
    }
