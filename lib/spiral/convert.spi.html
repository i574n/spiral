:= to' \* t u. (fn : t -> option u) (x : t) : option u =
    ~>
        ?: t * u ?|
        | ~x * ~x => Some `u x
        | int * int => Some `u (conv `t `u x)
        | float * float => Some `u (conv `t `u x)
        | int * float => Some `u (conv `t `u x)
        | float * int => Some `u (conv `t `u x)
        | i64 * f64 => Some `u (conv `t `u x)
        | f64 * i64 => Some `u (conv `t `u x)
        | i32 * f64 => Some `u (conv `t `u x)
        | f64 * i32 => Some `u (conv `t `u x)
        | i64 * f32 => Some `u (conv `t `u x)
        | f32 * i64 => Some `u (conv `t `u x)
        | char * bool => Some `u (x = '1')
        | bool * char => Some `u (? x ?> '1' |: '0')
        | bool * _ => Some `u (? x ?> 1 |: 0)
        | _ => fn x

:= lean_parse_helpers () : () =
    global """
        def spiralParseInt? (s : String) : Option Int :=
          let u := s.trimAscii.toString
          let v := if u.startsWith "+" then (u.drop 1).toString else u
          v.toInt?

        def spiralDigit (c : Char) : Int :=
          if c.isDigit then (c.toNat : Int) - 48
          else if 'a' ≤ c && c ≤ 'z' then (c.toNat : Int) - 87
          else if 'A' ≤ c && c ≤ 'Z' then (c.toNat : Int) - 55
          else 0

        def spiralParseIntBase (s : String) (base : Int32) : Int32 :=
          let u := s.trimAscii.toString
          let neg := u.startsWith "-"
          let body := if neg || u.startsWith "+" then (u.drop 1).toString else u
          let v := body.foldl (fun acc c => acc * base.toInt + spiralDigit c) 0
          Int32.ofInt (if neg then -v else v)
        """

:= convert_i32_base \* t. (base : i32) (x : t) : i32 =
    backend_switch {
        Gleam = \ () =>

            := (n : int), (err : int) =

            ?> n |> Some
            |: None
            |> optionm::value
        Lua = \ () => $'tonumber( !x, !base )' : i32
        Fsharp = \ () => $'System.Convert.ToInt32 (!x, !base)' : i32
        Python = \ () => $'int (!x, !base)' : i32
        Rust = \ () =>

        TypeScript = \ () => $'parseInt(!x, !base)' : i32
        C = \ () => $'(int32_t)strtol((!x)->ptr, NULL, !base)' : i32
        Pascal = \ () =>
            ? base = 16 ?> $'StrToInt(##36 + !x)' : i32
            |? base = 2 ?> $'StrToInt(##37 + !x)' : i32
            |? base = 8 ?> $'StrToInt(##38 + !x)' : i32

        Zig = \ () => $'(std.fmt.parseInt(i32, std.mem.trim(u8, !x, " \\t\\r\\n"), @@intCast(!base)) catch 0)' : i32

:= integer_fits \* u. (_ : u) (value : i64) : bool =
    := bounded : bool = ~> ?: u ?| i64 => !- | u64 => !- | _ => !+
    := lowest : i64 = ~> ?: u ?| i8 => -128i64 | i16 => -32768i64 | i32 => -2147483648i64 | _ => 0i64
    := highest : i64 =
        ~>
            ?: u ?|
            | i8 => 127i64
            | i16 => 32767i64
            | i32 => 2147483647i64
            | u8 => 255i64
            | u16 => 65535i64
            | u32 => 4294967295i64
            | _ => 0i64
    not bounded || (value >= lowest && value <= highest)

:= parsed_integer \* u. (ok : bool) (value : i64) : option u =
    := integral : bool =
        ~>
            ?: u ?|
            | i8 => !+
            | i16 => !+
            | i32 => !+
            | i64 => !+
            | u8 => !+
            | u16 => !+
            | u32 => !+
            | u64 => !+
            | _ => !-
    ? integral ?>
        := converted : u = ~> conv `i64 `u value
        ? ok && integer_fits converted value ?> Some converted |: None
    |: None

:= narrow_parsed \* u. (ok : bool) (value : f64) : option u =
    := low, high =
        ~>
            ?: u ?|
            | i8 => -128.0f64, 127.0f64
            | i16 => -32768.0f64, 32767.0f64
            | i32 => -2147483648.0f64, 2147483647.0f64
            | u8 => 0.0f64, 255.0f64
            | u16 => 0.0f64, 65535.0f64
            | _ => 0.0f64, 4294967295.0f64
    := converted : u = ~> conv `f64 `u value
    ? ok && value >= low && value <= high ?> Some converted |: None

:= try_convert \* t u. (x : t) : option u =
    x
    |> to' \ x =>
        backend_switch {
            Gleam = \ () =>
                ?
                    is_string_type x
                    && (
                        ~>
                            ?: u ?|
                            | i8 => !+
                            | i16 => !+
                            | i32 => !+
                            | i64 => !+
                            | u8 => !+
                            | u16 => !+
                            | u32 => !+
                            | u64 => !+
                            | _ => !-
                    )
                ?>
                    global "import gleam/int"
                    global "import gleam/string"
                    := (n : int), (err : int) =
                        $'case int.parse(string.trim(!x)) { Ok(n) -> ##(n, 0) Error(Nil) -> ##(0, 1) }'
                    ? err = 0
                    ?> n |> to |> Some
                    |: None : _ u
                |: x |> to |> Some
            Lua = \ () =>
                ? is_string_type x ?>
                    := ok = ($'(string.match(!x, "^%s*[+-]?%d+%s*$") ~= nil)' : bool)
                    := narrow : bool =
                        ~>
                            ?: u ?|
                            | i8 => !+
                            | i16 => !+
                            | i32 => !+
                            | u8 => !+
                            | u16 => !+
                            | u32 => !+
                            | _ => !-
                    ? narrow
                    ?> narrow_parsed ok ($'(tonumber(!x) or 0)' : f64)
                    |: parsed_integer ok ($'(tonumber(!x) or 0)' : i64)
                |: x |> to |> Some : _ u
            Fsharp = \ () =>
                resultm::from_try
                    \ () => $'!x |> `u '
                    eval
                |> resultm::ok
                : _ u
            Python = \ () =>
                resultm::from_try
                    \ () => $'`u(!x)'
                    eval
                |> resultm::ok
                : _ u
            Rust = \ () =>
                ? (~> ?: u ?| string => !+ | _ => !-)
                ?> Some $'Rc::<str>::from(format\!("{}", !x))'
                |? (~> ?: t ?| string => !+ | _ => !-) ?>
                    ($'(!x).trim().parse::<`u>().ok()' : optionm'::option' u) |> optionm'::unbox
                |: Some $'((!x) as `u)'
            C = \ () =>
                ? is_string_type x ?>
                    global "#include <stdbool.h>"
                    global "#include <stdlib.h>"
                    global "#include <ctype.h>"
                    global "#include <errno.h>"
                    global (
                        "static bool spiral_integer_text_ok(const char *s) { char *end; while (isspace((unsigned "
                        ^^ "char)*s)) s++; if (!*s) return false; errno = 0; strtoll(s, &end, 10); if (errno != 0 || "
                        ^^ "end == s) return false; while (isspace((unsigned char)*end)) end++; return *end == 0; }"
                    )
                    parsed_integer ($'spiral_integer_text_ok((!x)->ptr)' : bool) ($'strtoll((!x)->ptr, NULL, 10)' : i64)
                |: x |> to |> Some : _ u
            Pascal = \ () =>
                ? is_string_type x ?>
                    parsed_integer
                        ($'(StrToInt64Def(Trim(!x), 0) = StrToInt64Def(Trim(!x), 1))' : bool)
                        ($'StrToInt64Def(Trim(!x), 0)' : i64)
                |: x |> to |> Some : _ u
            TypeScript = \ () =>
                ? is_string_type x ?>
                    := ok : bool = $'/^\\s*[+-]?\\d+\\s*$/.test(!x)'
                    parsed_integer ok ($'(!ok ? BigInt(!x) : 0n)' : i64)
                |: x |> to |> Some : _ u
            Wasm = \ () =>
                ? is_string_type x ?>
                    parsed_integer ($'(call $integer_text_ok !x)' : bool) ($'(call $parse_i64 !x (i32.const 10))' : i64)
                |: x |> to |> Some : _ u
            Zig = \ () =>
                ? is_string_type x ?>
                    inl ok : bool = $'(if (std.fmt.parseInt(i64, std.mem.trim(u8, !x, " \\t\\r\\n"), 10)) |_| true \
                        else |_| false)'
                    parsed_integer ok ($'(std.fmt.parseInt(i64, std.mem.trim(u8, !x, " \\t\\r\\n"), 10) catch 0)' : i64)
                |? (~> ?: t ?| unativeint => !+ | _ => !-) || (~> ?: u ?| unativeint => !+ | _ => !-) ?>
                    Some ($'@@intCast(!x)' : u)
                |: x |> numeric_conv |> Some : _ u
            Lean = \ () =>
                ? is_string_type x ?>
                    lean_parse_helpers ()
                    parsed_integer ($'(spiralParseInt? !x).isSome' : bool) ($'(Int64.ofInt ((spiralParseInt? !x).getD \
                        0))' : i64)
                |: x |> to |> Some : _ u
            Kotlin = \ () =>
                ? is_string_type x ?>
                    parsed_integer
                        ($'((!x).trim().toLongOrNull() \!= null)' : bool)
                        ($'((!x).trim().toLongOrNull() ?: 0L)' : i64)
                |: x |> to |> Some : _ u
            Swift = \ () =>
                ? is_string_type x
                then parsed_integer ($'((!x).split(separator: " ").count == 1 && Int64(String((!x).split(separator: " \
                    ")[0])) \!= nil)' : bool) ($'(Int64(String((!x).split(separator: " ").first ?? "")) ?? 0)' : i64)
                |: x |> to |> Some : _ u
        }
