:= hex_nibble (c : string) : i32 = sm'::index_of c "0123456789abcdef"

:~ ^ reverse_bytes (xs : list u8) (acc : list u8) : list u8 =
    ?= xs ?|
    | [] => acc
    | x :: rest => reverse_bytes rest (x :: acc)

:~ ^ hex_bytes (text : string) (size : i32) (i : i32) (acc : list u8) : option (list u8) =
    ? i >= size ?> Some (reverse_bytes acc [])
    |:
        := hi = hex_nibble (sm'::slice i i text)
        := lo = hex_nibble (sm'::slice (i + 1i32) (i + 1i32) text)
        ? hi < 0i32 || lo < 0i32 ?> None
        |:
            := byte : u8 = convert::convert (hi * 16i32 + lo)
            hex_bytes text size (i + 2i32) (byte :: acc)

:= decode_hex (text : string) : option (list u8) =
    := size : i32 = sm'::length text
    ? size % 2i32 <> 0i32 ?> None |: hex_bytes text size 0i32 []

:= rust_ed25519_verify (key : array_base u8) (message : string) (signature : array_base u8) : bool =
    $'ed25519_dalek::VerifyingKey::from_bytes(&<[u8; 32]>::try_from(&(*!key.borrow())[..]).unwrap()) \
        .map(|k| k.verify_strict(!message.as_bytes(), &ed25519_dalek::Signature::from_bytes( \
            &<[u8; 64]>::try_from(&(*!signature.borrow())[..]).unwrap())).is_ok()) \
        .unwrap_or(false)'

:= ed25519_verify (public_key : string) (message : string) (signature : string) : bool =
    ?= decode_hex public_key ?|
    | None => !-
    | Some key =>
        ?= decode_hex signature ?|
        | None => !-
        | Some sig =>
            ? (listm::length key : i32) <> 32i32 || (listm::length sig : i32) <> 64i32 ?> !-
            |:
                := key_array = key |> listm'::box |> listm'::to_array'
                := sig_array = sig |> listm'::box |> listm'::to_array'
                rust_ed25519_verify key_array message sig_array
