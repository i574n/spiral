:= floor_div (a : i64) (b : i64) : i64 =
    := q = a / b
    ? a % b <> 0 && (a < 0) <> (b < 0) ?> q - 1 |: q

:= civil_era (days : i64) : i64 * i64 =
    := z = days + 719468
    := era = floor_div z 146097
    era, z - era * 146097

:= civil_year_of_era (day_of_era : i64) : i64 =
    (day_of_era - day_of_era / 1460 + day_of_era / 36524 - day_of_era / 146096) / 365

:= civil_day_of_year (day_of_era : i64) (year_of_era : i64) : i64 =
    day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100)

:= civil_month (month_from_march : i64) : i64 = ? month_from_march < 10 ?> month_from_march + 3 |: month_from_march - 9

:= civil_month_day (day_of_era : i64) (year_of_era : i64) : i64 * i64 =
    := day_of_year = civil_day_of_year day_of_era year_of_era
    := month_from_march = (5 * day_of_year + 2) / 153
    civil_month month_from_march, day_of_year - (153 * month_from_march + 2) / 5 + 1

:= civil_of_days (days : i64) : i64 * i64 * i64 =
    := era, day_of_era = civil_era days
    := year_of_era = civil_year_of_era day_of_era
    := month, day = civil_month_day day_of_era year_of_era
    year_of_era + era * 400 + (? month <= 2 ?> 1 |: 0), month, day

:= clock_of_day (of_day : i64) : i64 * i64 * i64 * i64 =
    := seconds = of_day / 1000000
    seconds / 3600, seconds % 3600 / 60, seconds % 60, of_day % 1000000

:= civil_of_micros (micros : i64) : i64 * i64 * i64 * i64 * i64 * i64 * i64 =
    := micros_per_day = 86400000000i64
    := days = floor_div micros micros_per_day
    := year, month, day = civil_of_days days
    := hour, minute, second, fraction = clock_of_day (micros - days * micros_per_day)
    year, month, day, hour, minute, second, fraction

:= span_fields (a : i64) : i64 * i64 * i64 * i64 * i64 =

:= day_prefix (days : i64) : string = ? days > 0i64 ?> "" ++#? days ++# "." |: ""

:= fraction_suffix (fraction : i64) : string = ? fraction > 0i64 ?> "." ++# sm'::pad_left 7 '0' ("" ++#? fraction) |: ""

:= clock_text (hours : i64) (minutes : i64) (seconds : i64) : string =
    span_two_digits hours ++# ":" ++# span_two_digits minutes ++# ":" ++# span_two_digits seconds

    := days, hours, minutes, seconds, fraction = fields
