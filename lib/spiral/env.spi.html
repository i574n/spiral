:= env_gleam_argv () =
    global "import gleam/erlang/charlist"
    global "import gleam/list"
    global "import gleam/result"

:= argument_offset () : i32 =
    run_target \|
        | TypeScript => \ () => 2i32
        | Gleam => \ () => 0i32
        | _ => \ () => 1i32

:= host_argument_count () : i32 =
    run_target \|
        | Rust => \ () => ($'std::env::args().count()' : convert::unativeint) |> convert::convert
        | Fsharp => \ () => $'System.Environment.GetCommandLineArgs().Length' : i32
        | TypeScript => \ () => $'process.argv.length' : i32
        | Cuda _ => \ () =>
            global "import sys"
            $'len(sys.argv)' : i32
        | Gleam => \ () =>
            env_gleam_argv ()
            $'list.length(spiral_env_plain_arguments())' : i32
        | target => \ () => "env.host_argument_count / target: " ++#? target |> failwith

:= host_argument (position : i32) : string =
    run_target_args (\ () => position) \|
        | Rust => \ position =>
            := position : convert::unativeint = convert::convert position
            $'Rc::<str>::from(std::env::args().nth(!position).unwrap_or_default())' : string
        | Fsharp => \ position => $'System.Environment.GetCommandLineArgs().[!position]' : string
        | TypeScript => \ position => $'process.argv[!position]' : string
        | Cuda _ => \ position =>
            global "import sys"
            $'sys.argv[!position]' : string
        | Gleam => \ position =>
            env_gleam_argv ()
            := rest = $'list.drop(spiral_env_plain_arguments(), !position)' : $'List(charlist.Charlist)'
            $'result.unwrap(result.map(list.first(!rest), charlist.to_string), "")' : string
        | target => \ _ => "env.host_argument / target: " ++#? target |> failwith

:= exit_with (code : i32) : () =
    run_target_args (\ () => code) \|
        | Rust => \ code => $'std::process::exit(!code)' : ()
        | Fsharp => \ code => $'exit !code' : ()
        | TypeScript => \ code => $'process.exit(!code)' : ()
        | Cuda _ => \ code =>
            global "import sys"
            $'sys.exit(!code)' : ()
        | target => \ _ => "env.exit_with / target: " ++#? target |> failwith

:~ command_line_argument_count () : i32 =
    host_argument_count () - argument_offset ()

:~ command_line_argument (index : i32) : string =
    index + argument_offset () |> host_argument

:= try_command_line_argument (index : i32) : option string =
    ? index >= 0 && index < command_line_argument_count ()

    |: None

:= argument_is_set (index : i32) : i32 =
    ?= index |> try_command_line_argument ?|
    | Some _ => 1i32
    | None => 0i32

:~ environment_snapshot () : string =
    run_target \|
        | Rust => \ () =>
            $'Rc::<str>::from(std::env::vars_os().map(|(key, value)| format\!("{}={}", key.to_string_lossy(), \
                value.to_string_lossy())).collect::<Vec<_>>().join(&char::from(0u8).to_string()).as_str())' : string
        | target => \ () => "env.environment_snapshot / target: " ++#? target |> failwith

:= environment_snapshot_usable (snapshot : string) : bool =
    := nul = sm'::char_text 0u8
    := held = nul ++# snapshot |> sm'::to_lower
    := keeps (name : string) : bool =
        get_environment_variable name = "" || held |> sm'::contains (nul ++# sm'::to_lower name ++# "=")
    := critical =
        ? platform::is_windows ()
        ?> [ "PATH"; "SystemRoot"; "TEMP"; "TMP"; "USERPROFILE"; "LOCALAPPDATA"; "APPDATA" ]
        |: [ "PATH"; "HOME" ]
    snapshot <> "" && (critical |> listm::fold (\ kept name => kept && keeps name) !+)

:= rust_remove_var (key : string) : () = $'unsafe { std::env::remove_var(&*!key) }' : ()

:== env_entries = array_base string

:= equals_after_first (entry : string) : i32 =
    := size : i32 = sm'::length entry
    ? size < 2i32 ?> -1i32 |: sm'::index_of "=" (sm'::substring 1i32 (size - 1i32) entry)

:= entry_pair (entry : string) : string * string =
    := found = equals_after_first entry
    := size : i32 = sm'::length entry
    ? found < 0i32 ?> "", ""
    |: sm'::substring 0i32 (found + 1i32) entry, sm'::substring (found + 2i32) (size - found - 2i32) entry

:= restorable (key : string) : bool = key <> "" && not (sm'::starts_with "=" key)

:~ ^ value_in (all : env_entries) (key : string) (i : i32) : option string =
    ? i >= am'::length_base all ?> None
    |:
        := k, v = entry_pair (am'::index_base i all)
        ? k = key && restorable k ?> Some v |: value_in all key (i + 1i32)

:= extra (wanted : env_entries) (kept : string) (key : string) : bool =
    restorable key && not (sm'::starts_with kept key) && value_in wanted key 0i32 = None

:= remove_if_extra (wanted : env_entries) (kept : string) (entry : string) : () =
    := key, _ = entry_pair entry
    ? extra wanted kept key ?> rust_remove_var key

:~ ^ removed_extras (current : env_entries) (wanted : env_entries) (kept : string) (i : i32) : () =
    ? i < am'::length_base current ?>
        remove_if_extra wanted kept (am'::index_base i current)
        removed_extras current wanted kept (i + 1i32)

:= set_if_changed (current : env_entries) (entry : string) : () =
    := key, value = entry_pair entry
    ? restorable key && value_in current key 0i32 <> Some value ?> rust_set_var key value

:~ ^ set_changed (wanted : env_entries) (current : env_entries) (i : i32) : () =
    ? i < am'::length_base wanted ?>
        set_if_changed current (am'::index_base i wanted)
        set_changed wanted current (i + 1i32)

:= nul_split (text : string) = sm'::split (sm'::char_text 0u8) text

:= rust_restored (snapshot : string) (kept_prefix : string) : () =
    := wanted = nul_split snapshot
    := current = nul_split (environment_snapshot ())
    removed_extras current wanted kept_prefix 0i32
    set_changed wanted current 0i32
