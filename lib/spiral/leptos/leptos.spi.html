:= element_to_fragment (view : view' (html_element _)) : fragment =
    view
    |> element_to_view
    |> view_to_fragment

:+ (~:>) fragment = \ x =>
    ~>
        ?: t ?|
        | array_base (view' (html_element ~el)) =>
            := x = array x
            := x = arraym::toList `array `(view' (html_element el)) x
            := x = listm::map `(view' (html_element el)) `view (element_to_view `el) x
            view_list_to_fragment x
        | list (view' (html_element ~el)) =>
            := x = listm::map `(view' (html_element el)) `view (element_to_view `el) x
            view_list_to_fragment x
        | list view =>
            view_list_to_fragment x
        | _ => x

:+ (~:>) view = \ x =>
    ~>
        ?: t ?|
        | view' (html_element _) => element_to_view x
        | _ => x

:= typecheck_signal \* (t : * -> *) u. (signal : t u) : () =
    ~>
        ?: t ?|
        | signal => ()
        | arc_signal => ()
        | rw_signal => ()
        | arc_rw_signal => ()
        | read_signal => ()
        | arc_read_signal => ()
        | write_signal => ()
        | arc_write_signal => ()
        | memo => ()
        | arc_memo => ()
        | _ => error_type `(()) ("invalid signal", ``(t u))

:= is_arc_memo \* (t : * -> *) u. (signal : t u) : bool =
    ~>
        ?: t ?|
        | arc_memo => !+
        | _ => !-

:= new_arc_memo \* t. (fn : () -> t) : arc_memo t =
    global """
        pub struct SpiralMemoChanged<'a, T>(pub Option<&'a T>, pub Option<&'a T>);
        pub trait SpiralMemoChangedEq { fn spiral_memo_changed(&self) -> bool; }
        impl<'a, T: PartialEq> SpiralMemoChangedEq for SpiralMemoChanged<'a, T> { fn spiral_memo_changed(&self) -> bool { self.0 != self.1 } }
        pub trait SpiralMemoChangedAny { fn spiral_memo_changed(&self) -> bool; }
        impl<'a, T> SpiralMemoChangedAny for &SpiralMemoChanged<'a, T> { fn spiral_memo_changed(&self) -> bool { true } }
        """
    := fn = ~~ fn
    := fn = (\ () => fn () |> send_wrapper_new) |> send_wrapper_new
    $'leptos::prelude::ArcMemo::new_with_compare(move |_| (*!fn)(), |a: \
        Option<&leptos::__reexports::send_wrapper::SendWrapper<`t>>, b: \
        Option<&leptos::__reexports::send_wrapper::SendWrapper<`t>>| (&SpiralMemoChanged(a.map(|x| &**x), b.map(|x| \
        &**x))).spiral_memo_changed())'

:= for \* t u (signal : * -> *).
    (signal : signal (am'::vec t))
    (key_fn : t -> u)
    (children' : t -> fragment)
    : view
    =
    signal |> typecheck_signal
    := items = (\ () => signal_get' signal : am'::vec t) |> send_wrapper_new
    := key = (\ (x : send_wrapper t) => x |> send_wrapper_get |> key_fn) |> send_wrapper_new
    := child = (\ (x : send_wrapper t) => x |> send_wrapper_get |> children' |> fragment_to_view) |> send_wrapper_new
    for [
        $'"each={ move || (*!items)().into_iter().map(leptos::__reexports::send_wrapper::SendWrapper::new).collect::<V\
            ec<_>>() }"'
        $'"key={ move |x: &leptos::__reexports::send_wrapper::SendWrapper<_>| (*!key)(x.clone()) }"'
        $'"children={ move |x: leptos::__reexports::send_wrapper::SendWrapper<_>| (*!child)(x) }"'
    ]

:= use_url () =
    := location = use_location ()

    \ () =>
        := url_pathname = location |> location_pathname |> signal_get |> sm'::from_std_string
        := url_search = location |> location_search |> signal_get |> sm'::from_std_string
        := url_search =
            ? url_search = ""
            ?> ""
            |: "?" ++# url_search
        url_pathname ++# url_search
    |> new_arc_memo
