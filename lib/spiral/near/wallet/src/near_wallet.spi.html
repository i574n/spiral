:< lustre

:== model = int

:= init initial_count : model =
    ? initial_count < 0
    ?> 0
    |: initial_count

:| msg =
    | Increment
    | Decrement

:= update n (model, msg) =
    ?= msg ?|
    | Increment => model + n
    | Decrement => model - n

:= view model =
    div [
        style [
            "display", "flex"
        ]
    ] [
        div [
            style [
                "display", "flex"
                "flex-direction", "column"
            ]
        ] [
            button [
                on_click Increment
            ] [
                text "+"
            ]
            p [
                style [
                    "text-align", "center"
                ]
            ] [
                model |> sm'::obj_to_string |> text
            ]
            button [
                on_click Decrement
            ] [
                text "-"
            ]
        ]
    ]
