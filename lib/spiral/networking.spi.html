:== request =
    {
        url : string
        request_type : request_type
        body : string
        json : bool
        auto_refresh : bool
    }

:= new_request_get (url : string) : request_builder =
    := url = ~~ url
    := url = url |> sm'::to_std_string
    := url = ~~ url
    !\($'"reqwest_wasm::Client::builder().build().map_err(|err| err.to_string())?.get(!url)"')

:= new_request_post (url : string) : request_builder =
    := url = ~~ url
    := url = url |> sm'::to_std_string
    := url = ~~ url
    !\($'"reqwest_wasm::Client::builder().build().map_err(|err| err.to_string())?.post(!url)"')

:= request_send (request : request_builder) : async::future_pin (resultm::result' reqwest_response reqwest_error) =
    := request = ~~ request
    !\($'"Box::pin(reqwest_wasm::RequestBuilder::send(!request))"')

:= request_body (body : string) (request : request_builder) : request_builder =
    := body = body |> sm'::to_std_string
    !\\(body, $'"reqwest_wasm::RequestBuilder::body(!request, $0)"')

:= request_header (key : string) (value : string) (request : request_builder) : request_builder =
    := request = ~~ request
    := key = key |> sm'::to_std_string
    := value = value |> sm'::to_std_string
    !\\((key, value), $'"reqwest_wasm::RequestBuilder::header(!request, $0, $1)"')

:= request_json \* t. (obj : t) (request : request_builder) : request_builder =
    !\($'"reqwest_wasm::RequestBuilder::json(!request, &!obj)"')

:= response_text (response : reqwest_response) : async::future_pin (resultm::result' sm'::std_string reqwest_error) =
    !\($'"Box::pin(reqwest_wasm::Response::text(!response))"')
