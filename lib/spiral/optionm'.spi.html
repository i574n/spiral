:= option'_types () : () =
    backend_switch {
        Gleam = \ () => global "import gleam/option"
    }
:! option'_gleam t =
    `(
        option'_types ()
        $'' : $'option.Option(`t)'
    )
:== option'_switch t =
    {
        Gleam : option'_gleam t
        Lua : t
        Fsharp : $"`t option"
        Python : t
        Rust : $'Option<`t>'
        TypeScript : t
        Zig : $'?`(t)'
        Lean : $'(Option `t)'
    }
:! option' t = $'backend_switch `(option'_switch t)'

:= none' \* t. () : option' t =
    backend_switch {
        Gleam = \ () => $'option.None' : option' t
        Lua = \ () => $'nil' : option' t
        Fsharp = \ () => $'None' : option' t
        Python = \ () => $'None' : option' t
        Rust = \ () => $'None' : option' t
        TypeScript = \ () => $'undefined' : option' t
        Zig = \ () => $'null' : option' t
        Lean = \ () => $'none' : option' t
    }

:= some' \* t. (x : t) : option' t =
    backend_switch {
        Gleam = \ () => $'!x |> option.Some' : option' t
        Lua = \ () => $'!x ' : option' t
        Fsharp = \ () => $'Some !x ' : option' t
        Python = \ () => $'!x ## some\' ' : option' t
        Rust = \ () => $'Some(!x.clone())' : option' t
        TypeScript = \ () => $'!x ' : option' t
        Zig = \ () => $'!x ' : option' t
        Lean = \ () => $'(some !x)' : option' t
    }

:= is_none' \* t. (x : option' t) : bool =
    backend_switch {
        Lua = \ () => $'!x == nil' : bool
        Python = \ () => $'!x is None' : bool
        Rust = \ () => $'!x.is_none()' : bool
        TypeScript = \ () => $'!x === undefined' : bool
    }

:= nullable_value' \* t. (x : option' t) : t =
    $'!x' : t

:= default_value' \* t. (value : t) (x : option' t) : t =
    backend_switch {
        Gleam = \ () => $'!x |> option.unwrap(!value)' : t
        Lua = \ () => ? is_none' x ?> value |: nullable_value' x
        Fsharp = \ () => $'!x |> Option.defaultValue !value ' : t
        Python = \ () => ? is_none' x ?> value |: nullable_value' x
        Rust = \ () =>
            := value = rust::clone value
            $'!x.unwrap_or(!value)' : t
        TypeScript = \ () => ? is_none' x ?> value |: nullable_value' x
        Zig = \ () => $'(!x orelse !value)' : t
        Lean = \ () => $'((!x).getD !value)' : t
    }

:= get' \* t. (x : option' t) : t =
    backend_switch {
        Gleam = \ () =>
            option'_types ()
            $'!x |> option.unwrap (Nil)' : t
        Lua = \ () => $'!x ' : t
        Fsharp = \ () => $'!x |> Option.get' : t
        Python = \ () => $'!x ' : t
        Rust = \ () => $'!x.unwrap()' : t
        TypeScript = \ () => $'!x ' : t
        Zig = \ () => $'!x.?' : t
        Lean = \ () => $'(!x).get\!' : t
    }

:= box \* t. (x : option t) : option' t =
    ?= x ?|
    | Some x => some' x
    | None => none' ()

:= map \* t u. (fn : t -> u) (x : option' t) : option' u =
    backend_switch {
        Gleam = \ () =>
            option'_types ()
            $'!x |> option.map(!fn)' : option' u
        Lua = \ () =>
            ? x =. none' ()
            ?> none' ()
            |: fn $'!x ' |> \ x => $'!x ' : option' u
        Fsharp = \ () =>
            := fn = ~~ fn
            $'!x |> Option.map !fn ' : option' u
        Python = \ () =>
            ? x =. none' ()
            ?> none' ()
            |: fn $'!x ' |> \ x => $'!x ' : option' u
        Rust = \ () =>
            := fn = rust::one_param_fn fn
            := x = ~~ x
            $'!x.map(|x| !fn(x))' : option' u
        TypeScript = \ () =>
            ? $'!x === undefined' : bool
            ?> none' ()
            |: fn $'!x ' |> \ x => $'!x ' : option' u
        Zig = \ () =>
            ? $'(!x == null)' : bool
            ?> none' ()
            |: x |> get' |> fn |> some'
        Lean = \ () =>
            ? $'(!x).isNone' : bool
            ?> none' ()
            |: x |> get' |> fn |> some'
    }

:= map'' \* t u. (fn : t -> u) (x : option' t) : option' u =
    x |> map fn

:= unbox \* t. (x : option' t) : option t =
    x |> map'' Some |> default_value' None
