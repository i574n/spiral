:= update_char (parser_state s) (c : char) : parser_state =
    ?= c ?|
    | '\n' =>
        { s ?|
            line_start = s::line_start + s::position::col
            position = { line = s::position::line + 1; col = 1 }
        }
    | _ =>
        { s ?|
            position = { line = s::position::line; col = s::position::col + 1 }
        }
    |> parser_state

:= update_span (t : string) (i : int) (len : int) (parser_state s' &= s) : parser_state =
    := endi = i + len
    :~ ^ 루프 (i' : int) (last_nl : int) (count : int) =
        ? i' >= endi
        ?> last_nl, count
        |:
            ?= t |> sm'::index i' ?|
            | '\n' => 루프 (i' + 1) i' (count + 1)
            | _ => 루프 (i' + 1) last_nl count
    := last_nl, n = 루프 i -1 0
    ? n = 0
    ?> { s' ?| position = { line = s'::position::line; col = s'::position::col + len } }
    |:
        { s' ?|
            line_start = last_nl + 1
            position =
                {
                    line = s'::position::line + n
                    col = endi - last_nl
                }
            text_length = s'::text_length
        }
    |> parser_state

:= update t s =

:= advance_while (fn : char -> bool) (t : string) (i : int) (parser_state s' &= s)
    : int * parser_state

    ? i >= s'::text_length
    ?> i, s

            ? i' > s'::text_length
            ?> s'::text_length
            |: i'
        := consumed = i' - i
        ? consumed = 0
        ?> i, s
