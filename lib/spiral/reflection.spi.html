:= styled_step (separator : string) (acc : list string) (c : char) : list string =
    := s = sm'::of_char c
    := lower = sm'::to_lower s
    ? lower <> s && listm::length acc > 0i32 ?> lower :: separator :: acc |: lower :: acc

:= styled_name (separator : string) (text : string) : string =
    sm'::to_char_list text |> listm::fold (styled_step separator) [] |> listm::rev |> sm'::concat_list ""

:= typed_name_snake \* t. ((typed_name text) : typed_name t) : string = styled_name "_" text

:= typed_name_kebab \* t. ((typed_name text) : typed_name t) : string = styled_name "-" text

:= typed_name_lower \* t. ((typed_name text) : typed_name t) : string = sm'::to_lower text
