:= from_try \* t. (fn : () -> t) (ex_fn : (() -> exn) -> exn) : result t exn =
    := ok x = x |> Ok : result t exn
    := error (e : exn) = e |> Error : result t exn
    backend_switch {
        Lua = \ () =>
            := caught (ex : exn) : option (result t exn) = Some (Error (ex_fn \ () => ex))
            try (\ () => Ok (fn ()) : result t exn) caught |> optionm::value
        Fsharp = \ () =>
            ~~ $'try !fn () |> !ok with ex -> (fun () -> ex) |> !ex_fn |> !error ' : result t exn
        Python = \ () =>
            $'fn = !fn '
            $'ok = !ok '
            $'error = !error '
            $'ex_fn = !ex_fn '
            $'try: x = ok(fn()) '
            $'except Exception as ex: x = error(ex_fn(lambda: ex))'
            $'x' : result t exn
        TypeScript = \ () =>
            $'(() => { try { return !ok(!fn()); } catch (ex) { return !error(!ex_fn(() => ex)); } })()' : result t exn
        Rust = \ () =>
            := caught (ex : exn) : option (result t exn) = Some (Error (ex_fn \ () => ex))
            try (\ () => Ok (fn ()) : result t exn) caught |> optionm::value
        Zig = \ () => fn () |> ok
        C = \ () => fn () |> ok
    }

:= unbox \* t u. (x : result' t u) : result t u =
    ? result'_union ()
    ?> $'!x ' : result t u
    |:
        := ok x : result t u = Ok x
        := error x : result t u = Error x
        := ok = ~~ ok
        := error = ~~ error
        backend_switch {
            Gleam = \ () =>
                $'case !x { Ok(x) -> !ok(x) Error(x) -> !error(x) }' : result t u

            Fsharp = \ () =>
                ~>
                    ?: t ?|
                    | () => $'match !x with Ok () -> !ok () | Error x -> !error x' : result t u

                    | () => $'match !x { Ok(()) => !ok(), Err(e) => !error(e) }' : result t u
