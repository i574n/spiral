    := ^ body acc = \|
        | StreamCons (st, fn') => 루프 (fn acc st) (fn' ())

:= to_list s =

        StreamCons (x, \ () => acc)

:| memoized_stream t =
    | NotComputed : () -> stream t
    | Computed : stream t

:= memoize s =
    := ^ body \* t. (s : () -> stream t) : () -> stream t =
        := state = mut (NotComputed s)
        \ () =>
            ?= *state ?|
            | Computed x => x
            | NotComputed fn =>
                := new_state =
                    ?= fn () ?|
                    | StreamNil => StreamNil
                    | StreamCons (x, fn) => StreamCons (x, 루프 fn)
                state <- Computed new_state
                new_state
    :& := 루프 \* t. (s' : () -> stream t) : () -> stream t =
        join_body_unit body s s'
    루프 \ () => s
