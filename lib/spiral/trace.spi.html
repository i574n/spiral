:= on_chain () : bool =
    run_target \|
        | Rust => \ () => $'cfg\!(target_arch = "wasm32")' : bool
        | _ => \ () => !-

:= read_state () =
    run_target \|
        | Zig => \ () =>
            {
                trace_level =
                    "TRACE_LEVEL" |> env::get_environment_variable |> reflection::union_try_pick : option trace_level
                repl_start = None : option i64
            }
        | Rust => \ () =>
            {
                trace_level =
                    ? on_chain ()
                    ?> None
                    |: "TRACE_LEVEL" |> env::get_environment_variable |> reflection::union_try_pick
                repl_start = None
            }
        | TypeScript => \ () =>
            {
                trace_level = "TRACE_LEVEL" |> env::get_environment_variable |> reflection::union_try_pick'
                repl_start = None
            }
        | _ => \ () =>
            ~~
                {
                    trace_level =
                        := get_result () =
                            "TRACE_LEVEL"
                            |> env::get_environment_variable
                            |> reflection::union_try_pick
                        backend_switch {
                            Gleam = get_result : () -> option trace_level
                            Lua = \ () => None : option trace_level
                            Fsharp = get_result : () -> option trace_level
                            Python = get_result : () -> option trace_level
                        }
                    repl_start =
                        := automation = env::get_environment_variable "AUTOMATION"
                        ? automation <>. "True"
                        ?> None
                        |:
                            date_time::now ()
                            |> date_time::to_timestamp
                            |> \ (date_time::timestamp x) => x
                            |> Some
                }

:~ new_trace_state trace_level' =
    := { repl_start trace_level } = read_state ()
    {
        count = mut 1i64
        trace_file = mut ignore
        enabled = mut !+
        acc = mut ""
        level = trace_level |> optionm'::default_value trace_level' |> mut
        repl_start = repl_start |> optionm'::box
    } : trace_state

:= init_trace_state trace_level : () =
    := trace_level = trace_level |> optionm'::default_value Verbose
    backend_switch {
        Gleam = \ () =>
            global "fn trace_state () { option.None }"
            $'let new_trace_state = !new_trace_state ' : ()
            $'let trace_level = !trace_level ' : ()
            $'let trace_state = fn () {' : ()
            $'  case trace_state () {' : ()
            $'    option.None -> new_trace_state(trace_level) |> option.Some' : ()
            $'    x -> x' : ()
            $'  }' : ()
            $'}' : ()
        Lua = \ () =>
            global "_G.trace_state = nil"
            \ () =>
                ? $'_G.trace_state == nil' : bool ?>
                    := trace_state = new_trace_state trace_level
                    $'_G.trace_state = !trace_state ' : ()
            |> exec_unit
        Fsharp = \ () =>
            backend_switch {
                Fsharp = \ () =>
                    global "module TraceState = let mutable trace_state = None"
            }
            \ () =>
                ? $'TraceState.trace_state.IsNone' ?>
                    := trace_state = new_trace_state trace_level |> optionm'::some'
                    $'TraceState.trace_state <- !trace_state ' : ()
            |> exec_unit
        Python = \ () =>
            global "class TraceState: trace_state = None"
            \ () =>
                ? $'TraceState.trace_state is None' : bool ?>
                    := trace_state = new_trace_state trace_level
                    $'TraceState.trace_state = !trace_state ' : ()
            |> exec_unit
        TypeScript = \ () =>
            global "let spiral_trace_state: any = undefined;"
            \ () =>
                ? $'spiral_trace_state === undefined' : bool ?>
                    := trace_state = new_trace_state trace_level
                    $'spiral_trace_state = !trace_state ' : ()
            |> exec_unit
        Rust = \ () =>
            global """
                fn spiral_trace_hold<T: Clone>(fresh: &std::rc::Rc<dyn Fn() -> T>) -> T {
                    use std::sync::OnceLock;
                    static SLOT: OnceLock<usize> = OnceLock::new();
                    let raw = *SLOT.get_or_init(|| Box::into_raw(Box::new((*fresh)())) as usize);
                    unsafe { (*(raw as *const T)).clone() }
                }
                """
            := fresh = \ () => new_trace_state trace_level
            :< rust_operators
            !\\(fresh, $'"{ let _ = spiral_trace_hold(&$0); }"') : ()
        Zig = \ () => ()
    }

:= get_trace_state_or_init trace_level : trace_state =
    init_trace_state trace_level
    backend_switch {
        Gleam = \ () =>
            := trace_level = trace_level |> optionm'::default_value Verbose
            $'trace_state () |> option.lazy_unwrap(fn () { !new_trace_state(!trace_level) })' : trace_state
        Lua = \ () =>
            $'_G.trace_state' : trace_state
        Fsharp = \ () =>
            $'TraceState.trace_state.Value' : trace_state
        Python = \ () =>
            $'TraceState.trace_state' : trace_state
        TypeScript = \ () =>
            $'spiral_trace_state' : trace_state
        Rust = \ () =>
            global """
                fn spiral_trace_hold<T: Clone>(fresh: &std::rc::Rc<dyn Fn() -> T>) -> T {
                    use std::sync::OnceLock;
                    static SLOT: OnceLock<usize> = OnceLock::new();
                    let raw = *SLOT.get_or_init(|| Box::into_raw(Box::new((*fresh)())) as usize);
                    unsafe { (*(raw as *const T)).clone() }
                }
                """
            := level' = trace_level |> optionm'::default_value Verbose
            := fresh = \ () => new_trace_state level'
            :< rust_operators
            !\\(fresh, $'"spiral_trace_hold(&$0)"') : trace_state
        Zig = \ () =>
            global """
                fn spiralHold(comptime T: type, fresh: anytype) T {
                    const S = struct {
                        var slot: ?T = null;
                    };
                    if (S.slot == null) S.slot = fresh.call(fresh.ctx);
                    return S.slot.?;
                }
                """
            := level' = trace_level |> optionm'::default_value Verbose
            := fresh = dyn \ () => new_trace_state level'
            $'spiralHold(`trace_state, !fresh)' : trace_state
    }

:= set_trace_level new_level =
    := ({ level } & trace_state) = get_trace_state_or_init None
    level <- new_level
    backend_switch {
        Gleam = \ () => $'let trace_state = fn () { !(trace_state |> optionm'.some') }' : ()
        Lua = id : () -> ()
        Fsharp = id : () -> ()
        Python = id : () -> ()
        Rust = id : () -> ()
        TypeScript = id : () -> ()
        Zig = id : () -> ()
    }

:= test_trace_level level : bool =
    := state = get_trace_state_or_init None
    := level' = *state::level
    := a : int = level |> severity
    := b : int = level' |> severity
    ? *state::enabled |> not
    ?> !-
    |: a >= b

                            #[cfg(target_arch = "wasm32")]
                            fn spiral_trace_near_log(text: &str) {
                                use std::sync::atomic::{AtomicUsize, Ordering};
                                static LOGGED: AtomicUsize = AtomicUsize::new(0);
                                let mut rest = text;
                                while !rest.is_empty() {
                                    let used = LOGGED.load(Ordering::Relaxed);
                                    let budget = 16000usize.saturating_sub(used);
                                    if budget < 13 {
                                        break;
                                    }
                                    let mut end = rest.len().min(15000);
                                    while !rest.is_char_boundary(end) {
                                        end -= 1;
                                    }
                                    let (chunk, tail) = rest.split_at(end);
                                    if chunk.len() <= budget {
                                        LOGGED.store(used + chunk.len(), Ordering::Relaxed);
                                        near_sdk::env::log_str(chunk);
                                        rest = tail;
                                    } else {
                                        let mut cut = budget - 12;
                                        while !chunk.is_char_boundary(cut) {
                                            cut -= 1;
                                        }
                                        let s = format!("{} [truncated]", &chunk[..cut]);
                                        LOGGED.store(used + s.len(), Ordering::Relaxed);
                                        near_sdk::env::log_str(&s);
                                        break;
                                    }
                                }
                            }
                            """
                        ? $'cfg\!(target_arch = "wasm32")' : bool
                        ?>

:= rust_clock_shims () : () =
    global """
        #[cfg(target_arch = "wasm32")]
        fn spiral_cfg_local_clock() -> Option<[i64; 3]> { None }
        #[cfg(all(windows, not(target_arch = "wasm32")))]
        fn spiral_cfg_local_clock() -> Option<[i64; 3]> {
            #[repr(C)] struct St([u16; 8]);
            unsafe extern "system" { fn GetLocalTime(t: *mut St); }
            let mut t = St([0; 8]);
            unsafe { GetLocalTime(&mut t) };
            Some([i64::from(t.0[4]), i64::from(t.0[5]), i64::from(t.0[6])])
        }
        #[cfg(all(unix, not(target_arch = "wasm32")))]
        fn spiral_cfg_local_clock() -> Option<[i64; 3]> {
            #[repr(C, align(8))] struct Tm([i32; 16]);
            unsafe extern "C" { fn localtime_r(t: *const std::os::raw::c_long, tm: *mut Tm) -> *mut Tm; }
            let now = std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).map(|d| d.as_secs());
            let secs = std::os::raw::c_long::try_from(now.unwrap_or(0)).unwrap_or(0);
            let mut tm = Tm([0; 16]);
            unsafe { localtime_r(&secs, &mut tm) };
            Some([i64::from(tm.0[2]), i64::from(tm.0[1]), i64::from(tm.0[0])])
        }
        #[cfg(not(any(windows, unix, target_arch = "wasm32")))]
        fn spiral_cfg_local_clock() -> Option<[i64; 3]> { None }
        #[cfg(target_arch = "wasm32")]
        fn spiral_cfg_epoch_nanos() -> u64 { near_sdk::env::block_timestamp() }
        #[cfg(not(target_arch = "wasm32"))]
        fn spiral_cfg_epoch_nanos() -> u64 {
            let since = std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH);
            since.map(|d| u64::try_from(d.as_nanos()).unwrap_or(0)).unwrap_or(0)
        }
        """

:= rust_local_clock () : optionm'::option' rust_clock_fields =
    rust_clock_shims ()
    $'spiral_cfg_local_clock()' : optionm'::option' rust_clock_fields

:= rust_epoch_nanos () : u64 =
    rust_clock_shims ()
    $'spiral_cfg_epoch_nanos()' : u64

:= rust_clock_hour (fields : rust_clock_fields) : i64 = $'!fields[0]' : i64

:= rust_clock_minute (fields : rust_clock_fields) : i64 = $'!fields[1]' : i64

:= rust_clock_second (fields : rust_clock_fields) : i64 = $'!fields[2]' : i64

:= local_seconds_of_day () : u64 =
    ?= rust_local_clock () |> optionm'::unbox ?|
    | Some fields =>
        conv (rust_clock_hour fields * 3600i64 + rust_clock_minute fields * 60i64 + rust_clock_second fields)
    | None => rust_epoch_nanos () / 1000000000u64 % 86400u64
