util (test)

test util: prelude 
///- --test static --symbol

:< testing
:| ^ ski =
    | I
    | K
    | S
    | App : ski * ski

:= ^ eval ski =
    ?= ski ?|
    | App (App (K, x), y) => x |> eval
    | App (App (App (S, x), y), z) => App (App (x, z), App (y, z)) |> eval
    | App (I, x) => x |> eval
    | App (K, x) => App (K, eval x)
    | App (f, x) => App (eval f, x) |> eval
    | _ => ski

ski (test)

|> _assert_eq I

__assert_eq { expected = I }
__assert_eq { expected = I }
