:= closure_wrap \* t. (x : rust::box t) : closure t =
    := x = ? is_rust () ?> x |: ~~ x
    !\($'"wasm_bindgen::closure::Closure::wrap(!x)"')

:= closure_forget \* t. (closure : closure t) =
    !\($'"!closure.forget()"') : ()

:= closure_as_ref \* t. (closure : closure t) : rust::ref js_value =
    !\($'"wasm_bindgen::closure::Closure::as_ref(&!closure)"')

:= unchecked_ref (ref : rust::ref js_value) : rust::ref js_function =
    !\($'"wasm_bindgen::JsCast::unchecked_ref(!ref)"')

:= set_inner_html (html : string) (el : html_element) =
    := html = ~~ html
    := html = html |> sm'::as_str
    := el = ~~ el
    !\\(html, $'"!el.set_inner_html($0)"')

:= from_js_value (value : js_value) : resultm::result' (optionm'::option' sm'::json_value) sm'::std_string =
    := value = ~~ value
    !\($'"serde_wasm_bindgen::from_value(!value)"')
    |> resultm::map_error' \ (x : sm'::serde_wasm_bindgen_error) => x |> sm'::format'
